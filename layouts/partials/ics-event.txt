{{- $p := .Params -}}
{{- /* Next date this quiz falls on */ -}}
{{- $next := now -}}
{{- range seq 0 6 -}}
  {{- $d := now.AddDate 0 0 . -}}
  {{- if eq ($d.Format "Monday") $p.day -}}{{- $next = $d -}}{{- break -}}{{- end -}}
{{- end -}}
{{- $start := printf "%sT%s00" ($next.Format "20060102") (replace $p.time ":" "") -}}
{{- $byday := index (dict "Monday" "MO" "Tuesday" "TU" "Wednesday" "WE" "Thursday" "TH" "Friday" "FR" "Saturday" "SA" "Sunday" "SU") $p.day -}}
{{- /* Clean every field: line breaks become spaces (blocks injection), then \ ; , are escaped as the calendar standard requires */ -}}
{{- $summary := printf "%s pub quiz" $p.pub | replaceRE "[\r\n]+" " " | replaceRE "([\\\\;,])" "\\$1" -}}
{{- $location := $p.address | replaceRE "[\r\n]+" " " | replaceRE "([\\\\;,])" "\\$1" -}}
{{- $desc := printf "Prizes: %s. Check with the pub before you go." $p.prize | replaceRE "[\r\n]+" " " | replaceRE "([\\\\;,])" "\\$1" -}}
{{- $url := (.OutputFormats.Get "html").Permalink -}}
